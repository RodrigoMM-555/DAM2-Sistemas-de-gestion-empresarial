usuarios
- id INTEGER PRIMARY KEY AUTOINCREMENT
- nombre VARCHAR(100) NOT NULL
- apellidos VARCHAR(150) NOT NULL
- email VARCHAR(254) NOT NULL UNIQUE
- telefono VARCHAR(20) NOT NULL UNIQUE
- rol ENUM('administrador', 'scouter', 'tutor') NOT NULL

educandos
- id INTEGER PRIMARY KEY AUTOINCREMENT
- nombre VARCHAR(100) NOT NULL
- apellidos VARCHAR(150) NOT NULL
- dni CHAR(9) NOT NULL UNIQUE
- fecha_nac DATE

educando_usuario
- id INTEGER PRIMARY KEY AUTOINCREMENT
- educando_id INTEGER NOT NULL REFERENCES educandos(id) ON DELETE CASCADE ON UPDATE CASCADE
- usuario_id INTEGER NOT NULL REFERENCES usuarios(id) ON DELETE CASCADE ON UPDATE CASCADE
- UNIQUE (educando_id, usuario_id)

eventos
- id INTEGER PRIMARY KEY AUTOINCREMENT
- nombre VARCHAR(150) NOT NULL
- responsable_id INTEGER REFERENCES usuarios(id) ON DELETE SET NULL ON UPDATE CASCADE
- fecha_inicio DATETIME NOT NULL
- fecha_final DATETIME NOT NULL
- lugar VARCHAR(150)
- municipio VARCHAR(100)
- provincia VARCHAR(100)
- tipo VARCHAR(50) NOT NULL

asistencias
- id INTEGER PRIMARY KEY AUTOINCREMENT
- evento_id INTEGER NOT NULL REFERENCES eventos(id) ON DELETE CASCADE
- educando_id INTEGER NOT NULL REFERENCES educandos(id) ON DELETE CASCADE
- asistencia ENUM('pendiente', 'si', 'no') NOT NULL DEFAULT 'pendiente'
- UNIQUE (evento_id, educando_id)
