categorias_factura
- codigo VARCHAR(50) PRIMARY KEY
- nombre VARCHAR(100) NOT NULL
- activo BOOLEAN NOT NULL DEFAULT TRUE
- orden SMALLINT UNSIGNED NOT NULL DEFAULT 0

tipos_evento
- codigo VARCHAR(50) PRIMARY KEY
- nombre VARCHAR(100) NOT NULL
- activo BOOLEAN NOT NULL DEFAULT TRUE
- orden SMALLINT UNSIGNED NOT NULL DEFAULT 0

secciones
- codigo VARCHAR(50) PRIMARY KEY
- nombre VARCHAR(100) NOT NULL
- edad_minima TINYINT UNSIGNED NULL
- edad_maxima TINYINT UNSIGNED NULL
- activo BOOLEAN NOT NULL DEFAULT TRUE
- orden SMALLINT UNSIGNED NOT NULL DEFAULT 0

usuarios
- nombre VARCHAR(100) NOT NULL
- apellidos VARCHAR(150) NOT NULL
- email VARCHAR(254) NOT NULL UNIQUE
- telefono VARCHAR(20) NOT NULL UNIQUE
- contraseña VARCHAR(255) NOT NULL
- rol ENUM('administrador', 'scouter', 'tutor') NOT NULL

educandos
- nombre VARCHAR(100) NOT NULL
- apellidos VARCHAR(150) NOT NULL
- DNI CHAR(9) NOT NULL UNIQUE
- naz DATE

facturas
- responsable_id INT UNSIGNED NOT NULL REFERENCES usuarios(id)
- numero_fact INT NOT NULL
- CIF_prov VARCHAR(20) NOT NULL
- fecha_emision DATETIME NOT NULL
- fecha_compra DATETIME NOT NULL
- concepto TEXT NOT NULL
- precio DECIMAL(10,2) NOT NULL
- categoria enum('envíos', 'comida', 'material', 'gasolina', 'otros', 'agua', 'luz', 'gas') NOT NULL
- estado ENUM('pendiente', 'pagada', 'rechazada') NOT NULL
  DEFAULT 'pendiente'
- imagen_factura VARCHAR(255) NULL

usuarios_educandos
- id_usuario INT UNSIGNED NOT NULL REFERENCES usuarios(id) ON DELETE CASCADE
- id_educando INT UNSIGNED NOT NULL REFERENCES educandos(id) ON DELETE CASCADE
- UNIQUE (id_usuario, id_educando)

eventos
- nombre VARCHAR(150) NOT NULL
- fecha_inicio DATETIME NOT NULL
- fecha_fin DATETIME NOT NULL
- id_responsable INT UNSIGNED REFERENCES usuarios(id) ON DELETE SET NULL
- lugar VARCHAR(150)
- municipio VARCHAR(100)
- comunidad VARCHAR(100)
- tipo VARCHAR(50) NOT NULL
- circular BOOLEAN NOT NULL DEFAULT FALSE

eventos_secciones
- id_evento INT UNSIGNED NOT NULL REFERENCES eventos(id) ON DELETE CASCADE
- seccion VARCHAR(50) NOT NULL
- PRIMARY KEY (id_evento, seccion)

educandos_asistencias
- id_educando INT UNSIGNED NOT NULL REFERENCES educandos(id) ON DELETE CASCADE
- id_evento INT UNSIGNED NOT NULL REFERENCES eventos(id) ON DELETE CASCADE
- estado ENUM('pendiente', 'si', 'no') NOT NULL DEFAULT 'pendiente'
- archivo_entrega VARCHAR(255)
- UNIQUE (id_educando, id_evento)
